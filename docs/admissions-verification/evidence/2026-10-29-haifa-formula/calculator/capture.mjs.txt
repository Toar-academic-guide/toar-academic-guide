let lastFinished = 0;
async function query(label, operation, params, localInput) {
  const path = `${root}/calculator/${label}.json`;
  if (fs.existsSync(path)) return JSON.parse(fs.readFileSync(path)).json;
  await page.waitForTimeout(Math.max(0, 10050 - (Date.now()-lastFinished)));
  const url = 'https://applicants.haifa.ac.il/enrollmentChances/CandChancesServlet?' + new URLSearchParams({operation,...params});
  const startedAt = new Date().toISOString();
  const response = await page.evaluate(async url => { const r = await fetch(url, {headers:{'X-Requested-With':'XMLHttpRequest'}}); return {status:r.status, body:await r.text()}; }, url);
  lastFinished = Date.now();
  const json = JSON.parse(response.body);
  fs.writeFileSync(path,JSON.stringify({label,operation,params,localInput,url,method:'GET',startedAt,finishedAt:new Date().toISOString(),status:response.status,json},null,2)+'\n');
  console.log(label, response.status, JSON.stringify(json).slice(0,200));
  return json;
}
